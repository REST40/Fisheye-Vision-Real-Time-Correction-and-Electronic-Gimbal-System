cmake_minimum_required(VERSION 3.20)
project(DoubleFisheyeHost VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network Test)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
find_path(OPENCV_INCLUDE_DIR opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
foreach(component core imgproc imgcodecs videoio)
  find_library(OPENCV_${component} NAMES opencv_${component} REQUIRED)
  list(APPEND OPENCV_LIBRARIES ${OPENCV_${component}})
endforeach()
find_library(MINIZIP_LIBRARY NAMES minizip REQUIRED)
add_library(host_core STATIC
  src/core/types.cpp src/core/projection.cpp src/core/npz_writer.cpp
  src/media/paired_source.cpp src/media/video_probe.cpp src/simulation/simulation_service.cpp)
target_include_directories(host_core PUBLIC src ${OPENCV_INCLUDE_DIR})
target_compile_definitions(host_core PRIVATE DF_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(host_core PUBLIC Qt6::Core Qt6::Gui ${OPENCV_LIBRARIES} ${MINIZIP_LIBRARY} ZLIB::ZLIB Threads::Threads)
file(GLOB BOARD_NETWORK_SOURCES CONFIGURE_DEPENDS src/fpga/network/*.cpp)
add_library(quality_reference STATIC
  src/quality/ps_reference/fptz_lut_generate.c src/quality/ps_reference/fptz_scene_mask.c
  src/quality/ps_reference/fptz_scene_mask_data.c)
set_target_properties(quality_reference PROPERTIES AUTOMOC OFF)
target_compile_options(quality_reference PRIVATE -ffp-contract=off -fno-fast-math)
add_library(host_ui STATIC ${BOARD_NETWORK_SOURCES}
  src/quality/qualityanalysis.cpp src/quality/qualityjob.cpp src/ui/imagequalitydialog.cpp src/ui/main_window.cpp src/ui/frameratepage.cpp src/ui/latencyanalysispage.cpp src/ui/settings_dialog.cpp src/ui/view_panel.cpp
  src/x5/capture_service.cpp src/fpga/fpga_link.h)
target_link_libraries(host_ui PUBLIC quality_reference host_core Qt6::Widgets Qt6::Network)
add_executable(double_fisheye_host src/app/main.cpp src/app/signal_shutdown.cpp)
target_link_libraries(double_fisheye_host PRIVATE host_ui)
foreach(target host_core host_ui double_fisheye_host)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
option(BUILD_X5_CAPTURE "Build the preserved, isolated X5 SDK/recorder executables" ON)
if(BUILD_X5_CAPTURE)
  set(CAMERA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/CameraSDK-2.1.1")
  set(MEDIA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/MediaSDK-3.1.1")
  foreach(sdk_file
      "${CAMERA_ROOT}/include/camera/camera.h"
      "${CAMERA_ROOT}/include/camera/device_discovery.h"
      "${CAMERA_ROOT}/lib/libCameraSDK.so"
      "${MEDIA_ROOT}/include/ins_realtime_stitcher.h"
      "${MEDIA_ROOT}/include/ins_stitcher.h"
      "${MEDIA_ROOT}/lib/libMediaSDK.so")
    if(NOT EXISTS "${sdk_file}")
      message(FATAL_ERROR "Missing X5 SDK file: ${sdk_file}\nInstall SDKs as described in dependencies/README.md, or configure with -DBUILD_X5_CAPTURE=OFF.")
    endif()
  endforeach()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale)
  pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
  add_executable(x5_recorder src/x5/recorder.cpp)
  target_link_libraries(x5_recorder PRIVATE PkgConfig::AV PkgConfig::SDL Threads::Threads)
  add_executable(x5_sdk_bridge src/x5/sdk_bridge.cpp)
  target_include_directories(x5_sdk_bridge PRIVATE "${CAMERA_ROOT}/include" "${MEDIA_ROOT}/include")
  target_link_libraries(x5_sdk_bridge PRIVATE "${CAMERA_ROOT}/lib/libCameraSDK.so" "${MEDIA_ROOT}/lib/libMediaSDK.so" Threads::Threads)
  set_target_properties(x5_sdk_bridge PROPERTIES BUILD_RPATH "${CAMERA_ROOT}/lib;${MEDIA_ROOT}/lib")
  foreach(name camera_probe exposure_probe)
    add_executable(x5_${name} src/x5/${name}.cpp)
    target_include_directories(x5_${name} PRIVATE "${CAMERA_ROOT}/include")
    target_link_libraries(x5_${name} PRIVATE "${CAMERA_ROOT}/lib/libCameraSDK.so" Threads::Threads)
    set_target_properties(x5_${name} PROPERTIES BUILD_RPATH "${CAMERA_ROOT}/lib")
  endforeach()
endif()
